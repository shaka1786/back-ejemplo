CREATE DATABASE IF NOT EXISTS GymDB3;
USE GymDB3;

-- =========================================
-- TABLA: Entrenador
-- =========================================
CREATE TABLE Entrenador (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) UNIQUE NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    horario VARCHAR(100)
);


-- =========================================
-- TABLA: Usuario
-- =========================================
CREATE TABLE Usuario (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo_electronico VARCHAR(100) UNIQUE NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    programa VARCHAR(100),
    rol VARCHAR(50),
    seguro VARCHAR(50),
    peso_inicial DECIMAL(5,2),
    tiempo_restante INT
);

-- =========================================
-- TABLA: Sesion
-- =========================================
CREATE TABLE Sesion (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_Entrenador INT,
    horario VARCHAR(100),
    FOREIGN KEY (id_Entrenador) REFERENCES Entrenador(id) ON DELETE CASCADE
);

-- =========================================
-- TABLA: TipoPagoMembresia (Renamed for clarity, since the other table is the relationship)
-- =========================================
CREATE TABLE TipoPagoMembresia (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tiempo INT NOT NULL,
    valor DECIMAL(10,2) NOT NULL
);

-- =========================================
-- RELACIÓN: Usuario - Sesion (Asiste)
-- =========================================
-- NOTE: Corrected the syntax error 'veces INT VARCHAR (1000)'. Assuming 'veces' should be an INT.
-- Also removed the trailing comma before the closing parenthesis.
CREATE TABLE AsistenciaUsuario (
    id_usuario INT,
    id_sesion INT, -- Added id_sesion to link to a specific session
    veces INT, -- Corrected data type to INT (or use VARCHAR(1000) if you must store a string)
    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_sesion), -- Composite primary key for the many-to-many relationship
    FOREIGN KEY (id_usuario) REFERENCES Usuario(id) ON DELETE CASCADE,
    FOREIGN KEY (id_sesion) REFERENCES Sesion(id) ON DELETE CASCADE
);


-- =========================================
-- RELACIÓN: Usuario - PagoMembresia (Realiza)
-- NOTE: The previous table definition was a duplicate of PagoMembresia. 
-- Renamed this relationship table to Usuario_Realiza_Pago to avoid conflict.
-- Assumed id_pago refers to the id in TipoPagoMembresia.
-- =========================================
CREATE TABLE Usuario_Realiza_Pago (
    id_usuario INT,
    id_pago INT,
    fecha_pago DATE DEFAULT (CURRENT_DATE),
    PRIMARY KEY (id_usuario, id_pago),
    FOREIGN KEY (id_usuario) REFERENCES Usuario(id) ON DELETE CASCADE,
    FOREIGN KEY (id_pago) REFERENCES TipoPagoMembresia(id) ON DELETE CASCADE
);
