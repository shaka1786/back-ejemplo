CREATE DATABASE IF NOT EXISTS GymDB;
USE GymDB;

-- =========================================
-- TABLA: Rol
-- =========================================
CREATE TABLE Rol (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);

-- =========================================
-- TABLA: HorarioLaboral
-- =========================================
CREATE TABLE HorarioLaboral (
    id INT AUTO_INCREMENT PRIMARY KEY,
    descripcion VARCHAR(100) NOT NULL UNIQUE
);

-- =========================================
-- TABLA: Usuario
-- =========================================
CREATE TABLE Usuario (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo_electronico VARCHAR(100) UNIQUE NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    programa VARCHAR(100),
    id_rol INT NOT NULL,
    id_horario_laboral INT NULL,
    seguro VARCHAR(50),
    peso_inicial DECIMAL(5,2),
    fecha_vencimiento INT,
    FOREIGN KEY (id_rol) REFERENCES Rol(id) ON DELETE RESTRICT,
    FOREIGN KEY (id_horario_laboral) REFERENCES HorarioLaboral(id) ON DELETE SET NULL
);

-- =========================================
-- TABLA: Sesion
-- (ahora referenciando Usuario en lugar de Entrenador)
-- =========================================
CREATE TABLE Sesion (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_entrenador INT NOT NULL,
    id_horario_plantilla INT NOT NULL,
    FOREIGN KEY (id_horario_plantilla) REFERENCES HorarioLaboral(id) ON DELETE CASCADE,
    FOREIGN KEY (id_entrenador) REFERENCES Usuario(id) ON DELETE CASCADE
);

-- =========================================
-- TABLA: TipoPagoMembresia
-- =========================================
CREATE TABLE TipoPagoMembresia (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tiempo INT NOT NULL,
    valor DECIMAL(10,2) NOT NULL
);

-- =========================================
-- RELACIÓN: Usuario - Sesion (Asiste)
-- =========================================
CREATE TABLE AsistenciaUsuario (
    id_usuario INT,
    id_sesion INT,
    veces INT,
    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_sesion),
    FOREIGN KEY (id_usuario) REFERENCES Usuario(id) ON DELETE CASCADE,
    FOREIGN KEY (id_sesion) REFERENCES Sesion(id) ON DELETE CASCADE
);

-- =========================================
-- RELACIÓN: Usuario - TipoPagoMembresia (Realiza)
-- =========================================
CREATE TABLE Usuario_Realiza_Pago (
    id_usuario INT,
    id_pago INT,
    fecha_pago DATE DEFAULT (CURRENT_DATE),
    PRIMARY KEY (id_usuario, id_pago),
    FOREIGN KEY (id_usuario) REFERENCES Usuario(id) ON DELETE CASCADE,
    FOREIGN KEY (id_pago) REFERENCES TipoPagoMembresia(id) ON DELETE CASCADE
);
